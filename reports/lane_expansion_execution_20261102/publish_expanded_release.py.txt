def api(method,url,payload=None,asset=None,name='api'):
    output=EXP/('expanded_'+name+'_response.json')
    cmd=['curl.exe','--silent','--show-error','--connect-timeout','30','--max-time','3600' if asset else '90','--config','-','--request',method,'--output',str(output),'--write-out','%{http_code}']
    if payload is not None:
        body=EXP/('expanded_'+name+'_request.json');body.write_text(json.dumps(payload,ensure_ascii=True),encoding='utf-8')
        cmd+=['--header','Content-Type: application/json','--data-binary','@'+str(body)]
    if asset:
        cmd+=['--header','Content-Type: application/octet-stream','--data-binary','@'+str(asset)]
    cmd.append(url)
    with (EXP/('expanded_'+name+'_transfer.log')).open('ab') as error:
        result=subprocess.run(cmd,input=secret_config,stdout=subprocess.PIPE,stderr=error)
    if result.returncode:raise RuntimeError(f'curl failed ({result.returncode}); see {name}_transfer.log')
    status=int(result.stdout.decode().strip())
    data=json.loads(output.read_bytes())
    if status>=400:raise RuntimeError(f'HTTP {status}: '+str(data.get('message',data)))
    return data
