from pathlib import Path
import json,time,hashlib
import pandas as pd
import pyarrow.parquet as pq
ROOT=Path(__file__).parent
while True:
    state=json.loads((ROOT/'run_status.json').read_text(encoding='utf-8'))
    if any(s['status']=='failed' for s in state['stages']):
        raise RuntimeError('Pipeline failed; inspect '+str(ROOT/(state['current_stage']+'.log')))
    if state.get('status')=='all_stages_completed':break
    time.sleep(30)
quality=ROOT/'continuity426-v1';dataset=ROOT/'pilot96-id-order-v1'

assert manifest['units']==96

for name in ['monthly_continuity','yearly_continuity','monthly_speed_drift','fold_continuity','gap_continuity','detector_identity_inventory']:
    pd.read_parquet(quality/(name+'.parquet')).to_csv(output/(name+'.csv'),index=False)
counts={}
for kind in ['bins','events']:
    destination=dataset/('cohort_'+kind+'.parquet');temporary=destination.with_suffix('.partial');writer=None;rows=0
    try:
        for source in sorted((dataset/'units').glob('*/'+kind+'.parquet')):
            checkpoint=json.loads((source.parent/'checkpoint.json').read_text(encoding='utf-8'))
            with source.open('rb') as stream:digest=hashlib.file_digest(stream,'sha256').hexdigest()
            assert digest==checkpoint['outputs_sha256'][source.name]
            table=pq.read_table(source)
            if writer is None:writer=pq.ParquetWriter(temporary,table.schema,compression='zstd')
            writer.write_table(table.cast(writer.schema));rows+=table.num_rows
    finally:
        if writer:writer.close()
    if writer is None:raise RuntimeError('No actual '+kind+' rows')
    temporary.replace(destination);counts[kind]=rows
assert counts['events']==manifest['events']
